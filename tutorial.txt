### Set Auto-Import ##

1. pnpm add unplugin-auto-import react-router-dom
2. open vite.config.ts
3. add this code:

```
import AutoImport from 'unplugin-auto-import/vite'

export default defineConfig({
  plugins: [
    ...
    AutoImport({
      imports: ['react', 'react-router-dom' ],
      // Auto import your components
      dirs: ["src/pages/**", 'src/api/**', 'src/hooks/**'],
      dts: 'src/auto-imports.d.ts',
    }),
    ....
})
```

## Set Routes Based-on Directory (like Next.js) ##

1. pnpm add vite-plugin-pages

2. open vite.config.ts, update code:

import { defineConfig } from "vite";
import react from "@vitejs/plugin-react";
import Pages from "vite-plugin-pages"; // Add this

export default defineConfig({
  plugins: [
    react(),
    // Add this
    Pages({
      dirs: "src/pages",
      extensions: ["tsx", "ts"],
    }),
  ],
});

3. wire routes on src/routes.tsx, add this code:

import { useRoutes } from "react-router-dom";
import routes from "~react-pages";

export default function Router() {
  return useRoutes(routes);
}

4. open main.tsx, update code:

import React from "react";
import ReactDOM from "react-dom/client";
import { BrowserRouter } from "react-router-dom";
import Router from "./routes"; // add this

ReactDOM.createRoot(document.getElementById("root")!).render(
  <React.StrictMode>
    <BrowserRouter>
      <Router /> // add this
    </BrowserRouter>
  </React.StrictMode>
);


5. example usage, create src/pages/about.tsx

export default function About() {
  return <h1>About</h1>;
}

6. example usage for dynamic route, create src/pages/blog/[slug].tsx

import { useParams } from "react-router-dom";

export default function BlogPost() {
  const { slug } = useParams();
  return <h1>Post: {slug}</h1>;
}

*note you can change "slug" wording with whatever you want
